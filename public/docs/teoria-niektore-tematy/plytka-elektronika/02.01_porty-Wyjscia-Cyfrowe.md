
# Monitor
```cpp
const byte pinOut = 4; 

void setup() {

  pinMode(pinOut, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  if (Serial.available()) {
    char ch = Serial.read();

    if (ch == '0') {
      digitalWrite(pinOut, LOW);
      // String + int → kompilator szuka operatora + dla String i int 
      // String ma zdefiniowany operator+ dla int → wynik: String("Pin: 4")
      // String + const char* → operator+ dla String i char* → wynik: String("Pin: 4 is LOW")
      Serial.println(String("Pin: ") + pinOut + " is LOW");
    }

    else if (ch == '1') {
      digitalWrite(pinOut, HIGH);
      Serial.println(String("Pin: ") + pinOut + " is HIGH");
    }
  }
}
```
