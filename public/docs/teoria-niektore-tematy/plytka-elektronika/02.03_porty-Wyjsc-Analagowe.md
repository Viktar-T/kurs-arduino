const byte outPin = 3;

void setup() {
  pinMode(outPin, OUTPUT);
  Serial.begin(9600);
  Serial.println("Wpisz napięcie w granicach od 0 V do 5 V");
}

void loop() {

  if (Serial.available() > 0) {
    float volts = Serial.parseFloat();
    while (Serial.available() > 0) Serial.read();
    int pwmValue = volts * 255 / 5;
    analogWrite(outPin, pwmValue);
  }
}
